{{- /* 通用链接解析：把配置/数据里写的路径，转成在任何部署环境都正确的地址 用法：{{ partial "url.html" "/index.xml" }} 处理规则： 1. 外链与特殊协议原样返回 （http://、https://、//、mailto:、tel:、data:、锚点 #） 2. 站内路径：去掉开头的 /，再交给 relLangURL —— 这一步是关键。Hugo 把带前导斜杠的路径当成「站点根绝对路径」， 不会拼接 baseURL 里的子路径，部署到 GitHub Pages 项目站点时会失效。 3. 空字符串或单独的 "/" → 返回当前语言首页 */ -}} {{- $src := . -}} {{- $out := $src -}} {{- if $src -}} {{- $s := strings.TrimSpace $src -}} {{- $external := or (hasPrefix $s "http://") (hasPrefix $s "https://") (hasPrefix $s "//") (hasPrefix $s "mailto:") (hasPrefix $s "tel:") (hasPrefix $s "data:") (hasPrefix $s "#") -}} {{- if $external -}} {{- $out = $s -}} {{- else -}} {{- $clean := strings.TrimPrefix "/" $s -}} {{- if eq $clean "" -}} {{- $out = site.Home.RelPermalink -}} {{- else -}} {{- $out = relLangURL $clean -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
